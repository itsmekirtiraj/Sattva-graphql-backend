Table users {
  id uuid [pk]
  phone_country_code varchar(5) [not null, default: '+91']
  phone_number varchar(15) [not null]
  phone_verified_at timestamp
  status account_status [not null, default: 'active']
  last_login_at timestamp
  created_at timestamp [not null]
  updated_at timestamp [not null]
  deleted_at timestamp

  indexes {
    (phone_country_code, phone_number) [unique]
  }

  Note: 'One account per normalized phone number. Store country code separately.'
}

  provider_reference varchar(255)

  indexes {
    (phone_country_code, phone_number, purpose, created_at)

  }

  Note: 'Never store OTP in plaintext. Limit attempts and resend frequency in the backend.'
}
